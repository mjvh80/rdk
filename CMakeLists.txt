cmake_minimum_required(VERSION 3.21)
project(rdk C)
enable_language(RC)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# FreeRDP 3.x (via vcpkg). Target/package names verified after install.
find_package(FreeRDP 3.26.0 EXACT CONFIG REQUIRED)
find_package(FreeRDP-Client CONFIG REQUIRED)
find_package(WinPR CONFIG REQUIRED)
include(cmake/Camera.cmake)
add_library(rdk_taskbar STATIC src/rdk_taskbar.cpp)
target_include_directories(rdk_taskbar PUBLIC src)
target_compile_features(rdk_taskbar PUBLIC cxx_std_17)
target_compile_definitions(rdk_taskbar PUBLIC UNICODE _UNICODE NOMINMAX)
target_link_libraries(rdk_taskbar PUBLIC shell32 propsys ole32 uuid user32)
add_executable(rdk-taskbar WIN32 src/rdk_taskbar_main.c)
target_link_libraries(rdk-taskbar PRIVATE rdk_taskbar)
add_library(rdk_clipboard_files STATIC src/rdk_clipboard_files.cpp)
target_include_directories(rdk_clipboard_files PUBLIC src)
target_compile_features(rdk_clipboard_files PUBLIC cxx_std_17)
target_compile_definitions(rdk_clipboard_files PUBLIC UNICODE _UNICODE NOMINMAX)
add_library(rdk_clipboard STATIC src/rdk_clipboard.cpp)
target_link_libraries(rdk_clipboard PUBLIC rdk_clipboard_files freerdp freerdp-client winpr ole32 shell32 uuid user32)

add_executable(rdk src/main.c src/rdk_gdi.c src/rdk_keyboard.c src/rdk_credentials.c src/rdk_capture.c src/rdk_pointer.c src/rdk_display.c)
target_sources(rdk PRIVATE src/rdk_media.c src/rdk_notice.c src/rdk_session.c src/rdk_startup.c src/rdk.rc src/rdk.manifest)
target_include_directories(rdk PRIVATE src assets)
target_link_libraries(rdk PRIVATE rdk_camera rdk_clipboard rdk_taskbar)
target_link_libraries(rdk PRIVATE freerdp freerdp-client winpr user32 gdi32 ws2_32 credui advapi32 ole32 uuid winmm)
target_compile_definitions(rdk PRIVATE UNICODE _UNICODE)
add_dependencies(rdk rdk-taskbar)
add_custom_command(TARGET rdk POST_BUILD
	COMMAND ${CMAKE_COMMAND} -E copy_if_different
		"${rdk_freerdp_SOURCE_DIR}/LICENSE" "$<TARGET_FILE_DIR:rdk>/FreeRDP-camera-LICENSE.txt")
